=====================================
buat table (users, categories, items)
=====================================
-------------------------------------
+users (id int auto_increment pk, name varchar(255), email varchar(255), password varchar(255))
--------------------------------------
CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(255) NOT null,
    email VARCHAR(255) NOT null,
    password VARCHAR(255) NOT null
);
-------------------------------------
- categories (id int auto_increment pk, name varchar(255))
-------------------------------------
CREATE TABLE categories (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(255) NOT null
);
-------------------------------------
- items (id int auto_increment pk, name varchar(255), description varchar(255), price int, stock int, category_id int fk categries (id))
-------------------------------------
CREATE TABLE items (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(255) NOT null,
    description VARCHAR(255) NOT null,
    price INT NOT null,
    stock INT NOT null,
    category_id INT NOT null,
    FOREIGN KEY (category_id) REFERENCES categories(id)
);

=====================================
Select data (users, categories, items)
=====================================

-------------------------------------
a. users (name)
-------------------------------------

-------------------------------------
b. items (all data diatas 1jt)
-------------------------------------
    SELECT * FROM items
    WHERE
        price > 1000001;

-------------------------------------
b. items (all data where data name ada kata Uniklo)
-------------------------------------
    SELECT * FROM items
    WHERE
        name like '%Uniklo%'

-------------------------------------
c. items + categories (all data by foreign key)
-------------------------------------
  SELECT items.id, items.name, items.price, items.stock, items.category_id, categories.name
    FROM items INNER JOIN categories
        ON items.category_id = categories.id;
